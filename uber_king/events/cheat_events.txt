popadanets_traits.0001 = {
    type = character_event
    title = popadanets.t
    desc = popadanets.d

    option = {
        name = popadanets.a
        custom_tooltip = popadanets.tooltip.add
        trigger = {
            NOR = {
                has_trait = popadanets_core_traits
            }
        }
        add_trait = popadanets_core_traits
    }

    option = {
        name = popadanets.b
        custom_tooltip = popadanets.tooltip.remove
        remove_trait = popadanets_core_traits
    }
}

cheat_combat_genius_realm.0001 = {
    type = character_event
    title = combat_genius_realm.t
    desc = combat_genius_realm.d

    option = {
        name = combat_genius_realm.a
        custom_tooltip = combat_genius_realm.tooltip.add
        trigger = {
            NOT = { has_realm_modifier = combat_genius_realm }
        }
        add_realm_modifier = {
            modifier = combat_genius_realm
            years = 100  # Or use 'duration = -1' for permanent
        }
    }

    option = {
        name = combat_genius_realm.b
        custom_tooltip = combat_genius_realm.tooltip.remove
        remove_realm_modifier = combat_genius_realm
    }
}
